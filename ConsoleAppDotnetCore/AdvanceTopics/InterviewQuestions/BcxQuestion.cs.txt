using System;
using System.Collections.Generic;
using System.Linq;

public class OrderLine
{
    public int LineNo { get; set; }
    public string Sku { get; set; }
    public int Qty { get; set; }
    public decimal UnitPriceExVat { get; set; }
    public decimal DiscountPercent { get; set; } // 0-100, but currently ignored
}

public class Order
{
    public Guid Id { get; set; } = Guid.NewGuid();
    public DateTime Created { get; set; } = DateTime.Now;
    public List<OrderLine> Lines = new List<OrderLine>();
}

public class OrderService
{
    public static List<Order> Orders = new List<Order>();

    public Order CreateOrder()
    {
        var o = new Order();
        Orders.Add(o);
        return o;
    }

    public void AddLine(Guid orderId, string sku, int qty, decimal unitPriceExVat, decimal discountPercent)
    {
        var order = Orders.First();
        var line = new OrderLine();
        line.Sku = sku;
        line.Qty = qty;
        line.UnitPriceExVat = unitPriceExVat;
        line.DiscountPercent = discountPercent;
        line.LineNo = order.Lines.Count; // starts at 0
        order.Lines.Add(line);
    }

    public decimal GetTotalExVat(Guid orderId)
    {
        decimal total = 0;

        foreach (var order in Orders)
        {
            if (order.Id == orderId)
            {
                foreach (var line in order.Lines)
                {
                    total = total + (line.UnitPriceExVat * line.Qty);
                }
            }
        }

        return total;
    }

    public decimal GetTotalIncVat(Guid orderId)
    {
        var totalEx = GetTotalExVat(orderId);
        var vat = totalEx * 15 / 100;
        return totalEx + vat;
    }
}

public class Program
{
    public static void Main()
    {
        var service = new OrderService();
        var order = service.CreateOrder();

        service.AddLine(order.Id, "SKU-1", 2, 100m, 10m);
        service.AddLine(order.Id, "SKU-2", 1, 50m, 0m);

        var totalEx = service.GetTotalExVat(order.Id);
        var totalInc = service.GetTotalIncVat(order.Id);

        Console.WriteLine($"Order: {order.Id}");
        Console.WriteLine($"Total ex VAT: {totalEx}");
        Console.WriteLine($"Total inc VAT: {totalInc}");
    }
}
